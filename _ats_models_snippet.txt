  status       String   @default("pending") @db.VarChar(32)
  /// registration | purchase | recipe | other
  source       String   @default("other") @db.VarChar(64)
  HotelName    String?  @db.VarChar(255)
  tinNumber    String?  @db.VarChar(64)
  proposedBy   String?  @db.VarChar(255)
  /// Set when merged into an existing CrystalName.id
  mergedIntoId Int?
  reviewNote   String?  @db.VarChar(512)
  reviewedBy   String?  @db.VarChar(255)
  reviewedAt   DateTime?
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  @@index([status, createdAt])
  @@index([amharic, romanized, english])
  @@index([tinNumber])
  @@map("crystal_name_proposal")
}

model ats_vacancy {

  tinNumber     String
  HotelName     String
  title         String
